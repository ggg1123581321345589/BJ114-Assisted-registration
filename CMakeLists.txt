cmake_minimum_required(VERSION 3.20)
project(AssistedRegistration VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Widgets Network)
find_package(CURL REQUIRED)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" config_upper)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config_upper} "${CMAKE_SOURCE_DIR}/bin")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/generated/main.moc"
  COMMAND Qt5::moc "${CMAKE_SOURCE_DIR}/src/main.cpp" -o "${CMAKE_BINARY_DIR}/generated/main.moc"
  DEPENDS src/main.cpp VERBATIM)
add_custom_target(query_moc DEPENDS "${CMAKE_BINARY_DIR}/generated/main.moc")
function(query_target target)
  target_include_directories(${target} PRIVATE src "${CMAKE_BINARY_DIR}/generated")
  target_link_libraries(${target} PRIVATE Qt5::Core Qt5::Widgets Qt5::Network CURL::libcurl)
  target_compile_definitions(${target} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /O2)
  endif()
endfunction()
add_executable(AssistedRegistration WIN32 src/main.cpp src/native_backend.cpp src/native_service.cpp)
add_dependencies(AssistedRegistration query_moc)
query_target(AssistedRegistration)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt5 REQUIRED COMPONENTS Test)
  add_executable(branding_probe tests/branding_probe.cpp src/native_backend.cpp src/native_service.cpp)
  add_dependencies(branding_probe query_moc)
  query_target(branding_probe)
  add_executable(native_ui_offline_test tests/native_ui_offline_test.cpp src/native_backend.cpp src/native_service.cpp)
  add_dependencies(native_ui_offline_test query_moc)
  query_target(native_ui_offline_test)
  target_link_libraries(native_ui_offline_test PRIVATE Qt5::Test)
  add_executable(native_service_offline_test tests/native_service_offline_test.cpp src/native_service.cpp)
  query_target(native_service_offline_test)
  add_executable(backend_offline_probe tests/backend_offline_probe.cpp)
  query_target(backend_offline_probe)
  add_test(NAME branding COMMAND branding_probe --assert-branding)
  add_test(NAME service_offline COMMAND native_service_offline_test)
  add_test(NAME ui_offline COMMAND native_ui_offline_test "${CMAKE_BINARY_DIR}/ui-offline-result.json")
  add_test(NAME backend_offline COMMAND backend_offline_probe "${CMAKE_SOURCE_DIR}")
  set_tests_properties(branding service_offline ui_offline backend_offline PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 120)
endif()
